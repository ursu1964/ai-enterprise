P5 — M4, Increment 5
Independent architecture review, revision lineage, approval authorization, and immutable approval evidence

This increment separates four distinct concepts:

Architecture generation
Architecture review
Architecture revision
Architecture approval

They must not collapse into one operation.

The governed flow becomes:

DRAFT architecture artifact
        ↓
submitted for independent review
        ↓
reviewer examines exact checksum
        ├── approve recommendation
        ├── request changes
        └── reject
        ↓
human architecture authority
        ├── approves exact reviewed checksum
        └── refuses approval

When changes are requested:

old artifact remains immutable
        ↓
new revision run created
        ↓
new artifact version generated
        ↓
review starts again

No artifact is reopened or edited in place.

9. Enforce separation of duties

Create:

app/architecture/governance_policy.py

from dataclasses import dataclass

from app.security.principal import Principal


@dataclass(frozen=True, slots=True)
class ArchitectureGovernancePolicy:
    version: str = "architecture-governance-v1"

    require_distinct_reviewer_and_approver: bool = True
    prohibit_generator_as_reviewer: bool = True
    prohibit_reviewer_as_approver: bool = True

    def assert_can_review(
        self,
        *,
        principal: Principal,
        generator_identity: str | None,
    ) -> None:
        if not principal.has_permission(
            "architecture.review.complete"
        ):
            raise PermissionError(
                "Principal cannot complete architecture reviews."
            )

        if (
            self.prohibit_generator_as_reviewer
            and generator_identity is not None
            and principal.subject_id == generator_identity
        ):
            raise PermissionError(
                "Architecture generator cannot review its own output."
            )

    def assert_can_approve(
        self,
        *,
        principal: Principal,
        reviewer_identity: str,
    ) -> None:
        if not principal.has_permission(
            "architecture.approve"
        ):
            raise PermissionError(
                "Principal cannot approve architecture."
            )

        if (
            self.prohibit_reviewer_as_approver
            and principal.subject_id == reviewer_identity
        ):
            raise PermissionError(
                "Architecture reviewer cannot approve the same artifact."
            )

    async def next_review_round(
        self,
        artifact_id: uuid.UUID,
    ) -> int:
        result = await self._session.execute(
            select(
                func.coalesce(
                    func.max(
                        ArchitectureReview.review_round
                    ),
                    0,
                )
            ).where(
                ArchitectureReview.architecture_artifact_id
                == artifact_id
            )
        )

        return int(result.scalar_one()) + 1

    async def get_open_review(
        self,
        artifact_id: uuid.UUID,
    ) -> ArchitectureReview | None:
        result = await self._session.execute(
            select(ArchitectureReview)
            .where(
                ArchitectureReview.architecture_artifact_id
                == artifact_id,
                ArchitectureReview.status
                == ArchitectureReviewStatus.OPEN.value,
            )
            .order_by(
                ArchitectureReview.review_round.desc()
            )
            .limit(1)
        )

        return result.scalar_one_or_none()

    async def get_review(
        self,
        review_id: uuid.UUID,
        *,
        for_update: bool = False,
    ) -> ArchitectureReview | None:
        statement = select(
            ArchitectureReview
        ).where(
            ArchitectureReview.id == review_id
        )

        if for_update:
            statement = statement.with_for_update()

        result = await self._session.execute(statement)
        return result.scalar_one_or_none()

    async def add_revision_request(
        self,
        revision: ArchitectureRevisionRequest,
    ) -> ArchitectureRevisionRequest:
        self._session.add(revision)
        await self._session.flush()
        return revision

    async def get_revision_request(
        self,
        revision_request_id: uuid.UUID,
        *,

    ) -> ArchitectureRevisionRequest | None:
        statement = select(

            == revision_request_id
        )

        if for_update:
            statement = statement.with_for_update()

        result = await self._session.execute(statement)
        return result.scalar_one_or_none()

15. Complete a review

Add:

    async def complete_review(
        self,
        command: CompleteArchitectureReviewCommand,
    ) -> ArchitectureReview:
        review = await self._repository.get_review(
            command.review_id,
            for_update=True,
        )

        if review is None:
            raise ArchitectureNotFoundError(
                f"Architecture review not found: "
                f"{command.review_id}."
            )

        if (
            review.status
            != ArchitectureReviewStatus.OPEN.value
        ):
            raise InvalidArchitectureTransitionError(
                "Only an open review may be completed."
            )

        if review.reviewer_id != command.completed_by:
            raise ArchitectureApprovalNotAllowedError(
                "Only the assigned reviewer may complete "
                "the review."
            )

        artifact = await self._repository.get_artifact(
            review.architecture_artifact_id,
            for_update=True,
        )

        if artifact is None:
            raise ArchitectureArtifactNotFoundError(
                "Architecture artifact not found."
            )

        recalculated_checksum = architecture_checksum(
            markdown_content=artifact.markdown_content,
            structured_content=artifact.structured_content,
        )

        if recalculated_checksum != artifact.content_checksum:
            raise ArchitectureChecksumMismatchError(
                "Stored architecture content no longer matches "
                "its checksum."
            )

        if review.reviewed_checksum != artifact.content_checksum:
            raise ArchitectureChecksumMismatchError(
                "Review was opened against a different artifact "
                "checksum."
            )

        if (
            command.decision
            == ArchitectureReviewDecision
            .RECOMMEND_APPROVAL
        ):
            blocking_findings = [
                finding
                for finding in command.findings
                if finding.get("blocking") is True
            ]

            if blocking_findings:
                raise ArchitectureApprovalNotAllowedError(
                    "A review with blocking findings cannot "
                    "recommend approval."
                )

            artifact.status = (
                ArchitectureArtifactStatus.UNDER_REVIEW.value
            )

        elif (
            command.decision
            == ArchitectureReviewDecision
            .REQUEST_CHANGES
        ):
            artifact.status = (
                ArchitectureArtifactStatus
                .CHANGES_REQUESTED.value
            )

        elif (
            command.decision
            == ArchitectureReviewDecision.REJECT
        ):
            artifact.status = (
                ArchitectureArtifactStatus.REJECTED.value
            )

        review.decision = command.decision.value
        review.comments = command.comments
        review.findings = command.findings
        review.status = ArchitectureReviewStatus.COMPLETED.value
        review.completed_at = datetime.now(UTC)

        event_type = {
            ArchitectureReviewDecision.RECOMMEND_APPROVAL:
                "architecture.review.recommended_approval",
            ArchitectureReviewDecision.REQUEST_CHANGES:
                "architecture.review.changes_requested",
            ArchitectureReviewDecision.REJECT:
                "architecture.review.rejected",
        }[command.decision]

        await self._audit_writer.append(
            AuditRecord(
                event_type=event_type,
                aggregate_type="architecture_review",
                aggregate_id=review.id,
                project_id=artifact.project_id,
                payload={
                    "artifact_id": str(artifact.id),
                    "artifact_version": artifact.version,
                    "decision": review.decision,
                    "finding_count": len(review.findings),
                    "reviewed_checksum":
                        review.reviewed_checksum,
                },
            )
        )

        await self._session.commit()
        await self._session.refresh(review)

        return review

@dataclass(frozen=True, slots=True)
class ApprovedArchitectureSnapshot:
    artifact_id: str
    project_id: str
    version: int
    checksum: str
    requirements_artifact_id: str
    approval_id: str
    approving_review_id: str
    approval_evidence_checksum: str


def assert_work_package_eligible(
    *,
    artifact,
    approval,
    review,
) -> ApprovedArchitectureSnapshot:
    if (
        artifact.status
        != ArchitectureArtifactStatus.APPROVED.value
    ):
        raise ArchitectureApprovalNotAllowedError(
            "Work packages require approved architecture."
        )

    recalculated = architecture_checksum(
        markdown_content=artifact.markdown_content,
        structured_content=artifact.structured_content,
    )

    if recalculated != artifact.content_checksum:
        raise ArchitectureApprovalNotAllowedError(
            "Architecture checksum verification failed."
        )

    if approval.approved_checksum != artifact.content_checksum:
        raise ArchitectureApprovalNotAllowedError(
            "Approval checksum does not match architecture."
        )

    if review.reviewed_checksum != artifact.content_checksum:
        raise ArchitectureApprovalNotAllowedError(
            "Review checksum does not match architecture."
        )

    return ApprovedArchitectureSnapshot(
        artifact_id=str(artifact.id),
        project_id=str(artifact.project_id),
        version=artifact.version,
        checksum=artifact.content_checksum,
        requirements_artifact_id=str(
            artifact.requirements_artifact_id
        ),
        approval_id=str(approval.id),
        approving_review_id=str(review.id),
        approval_evidence_checksum=(
            approval.approval_evidence_checksum
        ),
    )

from collections import deque
from dataclasses import dataclass
from typing import Any

from app.architecture.execution_result import (
    InvalidArchitectureExecution,
    ValidArchitectureExecution,
)


@dataclass(slots=True)
class RuntimeCall:
    operation: str
    input_checksum: str
    payload: dict[str, Any]


class ScriptedArchitectureRuntime:
    def __init__(
        self,
        scripted_results: list[
            ValidArchitectureExecution
            | InvalidArchitectureExecution
            | Exception
        ],
    ) -> None:
        self._results = deque(scripted_results)
        self.calls: list[RuntimeCall] = []

    async def generate(
        self,
        *,
        crew_input,
        approved_requirement_ids,
    ):
        self.calls.append(
            RuntimeCall(
                operation="generate",
                input_checksum="",
                payload={
                    "project_id": str(
                        crew_input.project_id
                    ),
                    "requirement_ids": sorted(
                        approved_requirement_ids
                    ),
                },
            )
        )

        return self._next_result()

    async def repair(
        self,
        *,
        crew_input,
        approved_requirement_ids,
        invalid_output,
        validation_report,
    ):
        self.calls.append(
            RuntimeCall(
                operation="repair",
                input_checksum="",
                payload={
                    "project_id": str(
                        crew_input.project_id
                    ),
                    "requirement_ids": sorted(
                        approved_requirement_ids
                    ),
                    "invalid_output": invalid_output,
                    "validation_report":
                        validation_report,
                },
            )
        )

        return self._next_result()

    def _next_result(self):
        if not self._results:
            raise AssertionError(
                "Scripted runtime received an unexpected call."
            )

        result = self._results.popleft()

        if isinstance(result, Exception):
            raise result

        return result

This lets each test declare exactly how the model runtime behaves.

    ):
        await harness.execute_run(
            run_id=run.id
        )

    attempts = await harness.get_attempts(run.id)

    assert len(attempts) == 2
    assert all(
        attempt.status == "timed_out"
        for attempt in attempts

P5 — M5, Increment 1
Work-package aggregate, persistence model, state machine, and architecture eligibility boundary

M5 converts one approved architecture into bounded, reviewable units of implementation work.

The target flow is:

Approved Architecture
        ↓
Work-Package Decomposition Run
        ↓
Candidate Work Packages
        ↓
Deterministic Validation
        ↓
Dependency Graph
        ↓
Human Review and Approval
        ↓
Execution Eligibility

A work package is not a free-form task description. It is an immutable execution contract.
1. Work-package invariants

The system must enforce these rules:

1. Work packages can be generated only from an approved architecture.

2. The architecture approval, review, artifact checksum, and project
   lineage must all match.

3. Every work package belongs to exactly one decomposition run.

4. Every package references one approved architecture version.

5. Package scope must be explicit and bounded.

6. Package file access must be expressed as allowlists and denylists.

7. Package test commands must be explicit and machine-executable.

8. Package dependencies must reference packages in the same decomposition.

9. The dependency graph must be acyclic.

10. Every architecture requirement and module assigned to implementation
    must be traceable to one or more packages.

11. A package cannot approve itself.

12. Approval is bound to the exact package checksum.

13. Approved packages are immutable.

14. Changes require a new package version or a new decomposition run.

15. Execution starts only from an approved package and exact repository base.

import hashlib
import json
from typing import Any


def canonical_json(
    value: Any,
) -> str:
    return json.dumps(
        value,
        sort_keys=True,
        separators=(",", ":"),
        ensure_ascii=False,
    )


def sha256_text(
    value: str,
) -> str:
    return hashlib.sha256(
        value.encode("utf-8")
    ).hexdigest()


def work_package_checksum(
    *,
    markdown_content: str,
    structured_content: dict[str, Any],
) -> str:
    payload = {
        "markdown_content": markdown_content,
        "structured_content": structured_content,
    }

    return sha256_text(
        canonical_json(payload)
    )


def approval_evidence_checksum(
    evidence: dict[str, Any],
) -> str:
    return sha256_text(
        canonical_json(evidence)
    )

class ApprovedArchitectureGateway:
    def __init__(
        self,
        session: AsyncSession,
    ) -> None:
        self._session = session

    async def get_approved_snapshot(
        self,
        *,
        architecture_artifact_id: uuid.UUID,
        architecture_approval_id: uuid.UUID,
    ) -> ApprovedArchitectureSnapshot | None:
        result = await self._session.execute(
            select(
                ArchitectureArtifact.id,
                ArchitectureArtifact.project_id,
                ArchitectureArtifact.requirements_artifact_id,
                ArchitectureArtifact.version,
                ArchitectureArtifact.content_checksum,
                ArchitectureArtifact.markdown_content,
                ArchitectureArtifact.structured_content,
                ArchitectureApproval.id.label(
                    "approval_id"
                ),
                ArchitectureApproval.evidence_checksum,
                ArchitectureReview.id.label(
                    "review_id"
                ),
                ArchitectureReview.reviewed_checksum,
            )
            .join(
                ArchitectureApproval,
                ArchitectureApproval.architecture_artifact_id
                == ArchitectureArtifact.id,
            )
            .join(
                ArchitectureReview,
                ArchitectureReview.id
                == ArchitectureApproval.approving_review_id,
            )
            .where(
                ArchitectureArtifact.id
                == architecture_artifact_id,
                ArchitectureApproval.id
                == architecture_approval_id,
                ArchitectureApproval.approved.is_(True),
                ArchitectureApproval.approved_checksum
                == ArchitectureArtifact.content_checksum,
                ArchitectureReview.reviewed_checksum
                == ArchitectureArtifact.content_checksum,
            )
        )

        row = result.one_or_none()

        if row is None:
            return None

        return ApprovedArchitectureSnapshot(
            artifact_id=row.id,
            project_id=row.project_id,
            requirements_artifact_id=(
                row.requirements_artifact_id
            ),
            version=row.version,
            checksum=row.content_checksum,
            markdown_content=row.markdown_content,
            structured_content=row.structured_content,
            approval_id=row.approval_id,
            approval_evidence_checksum=(
                row.evidence_checksum
            ),
            review_id=row.review_id,
            review_checksum=row.reviewed_checksum,

class WorkPackageService:
    def __init__(
        self,
        *,
        session: AsyncSession,
        repository: WorkPackageRepository,
        architecture_gateway: ApprovedArchitectureGateway,
        audit_writer,
    ) -> None:
        self._session = session
        self._repository = repository
        self._architecture_gateway = architecture_gateway
        self._audit_writer = audit_writer

    async def create_decomposition_run(
        self,
        command: CreateDecompositionRunCommand,
    ) -> WorkPackageDecompositionRun:
        architecture = (
            await self._architecture_gateway
            .get_approved_snapshot(
                architecture_artifact_id=(
                    command.architecture_artifact_id
                ),
                architecture_approval_id=(
                    command.architecture_approval_id
                ),
            )
        )

        if architecture is None:
            raise ApprovedArchitectureRequiredError(
                "Work-package decomposition requires an "
                "approved architecture and matching approval."
            )

        if architecture.project_id != command.project_id:
            raise ArchitectureProjectMismatchError(
                "Architecture does not belong to the "
                "requested project."
            )

        active_run = await self._repository.find_active_run(
            project_id=command.project_id,
            architecture_artifact_id=(
                command.architecture_artifact_id
            ),
        )

        if active_run is not None:
            raise ActiveDecompositionRunExistsError(
                "An active work-package decomposition run "
                "already exists for this architecture."
            )

        run = WorkPackageDecompositionRun(
            project_id=command.project_id,
            architecture_artifact_id=(
                architecture.artifact_id
            ),
            architecture_approval_id=(
                architecture.approval_id
            ),
            architecture_version=architecture.version,
            architecture_checksum=architecture.checksum,
            requirements_artifact_id=(
                architecture.requirements_artifact_id
            ),
            status=(
                WorkPackageDecompositionRunStatus
                .PENDING.value
            ),
            decomposition_version=(
                command.decomposition_version
            ),
            schema_version=command.schema_version,
            model_name=command.model_name,
            temperature=command.temperature,
            prompt_bundle_hash=(
                command.prompt_bundle_hash
            ),
            execution_manifest={
                **command.execution_manifest,
                "architecture_review_id": str(
                    architecture.review_id
                ),
                "architecture_review_checksum":
                    architecture.review_checksum,
                "architecture_approval_evidence_checksum":
                    architecture.approval_evidence_checksum,
            },
        )

        try:
            await self._repository.add_run(run)

            await self._audit_writer.append(
                {
                    "event_type":
                        "work_packages.decomposition.created",
                    "aggregate_type":
                        "work_package_decomposition_run",
                    "aggregate_id": run.id,
                    "project_id": run.project_id,
                    "payload": {
                        "architecture_artifact_id": str(
                            run.architecture_artifact_id
                        ),
                        "architecture_approval_id": str(
                            run.architecture_approval_id
                        ),
                        "architecture_version":
                            run.architecture_version,
                        "architecture_checksum":
                            run.architecture_checksum,
                        "schema_version":
                            run.schema_version,
                    },
                }
            )

            await self._session.commit()
        except IntegrityError as exc:
            await self._session.rollback()

            raise ActiveDecompositionRunExistsError(
                "A decomposition run was created "
                "concurrently."
            ) from exc

        await self._session.refresh(run)
        return run

    async def start_decomposition_run(
        self,
        command: StartDecompositionRunCommand,
    ) -> WorkPackageDecompositionRun:
        run = await self._repository.get_run(
            command.run_id,
            for_update=True,
        )

        if run is None:
            raise WorkPackageRunNotFoundError(
                f"Decomposition run not found: "
                f"{command.run_id}."
            )

        current = WorkPackageDecompositionRunStatus(
            run.status
        )

        target = (
            WorkPackageDecompositionRunStatus.RUNNING
        )

        assert_run_transition(current, target)

        run.status = target.value
        run.started_at = datetime.now(UTC)

        await self._session.commit()
        await self._session.refresh(run)

        return run

from dataclasses import dataclass, field

from app.work_packages.artifact_schema import (
    WorkPackageDecompositionDocument,
)
from app.work_packages.exceptions import (

@dataclass(slots=True)
class WorkPackageValidationFinding:
    code: str
    message: str
    location: str
    severity: str = "error"


@dataclass(slots=True)
class WorkPackageValidationReport:
    valid: bool
    findings: list[
        WorkPackageValidationFinding
    ] = field(default_factory=list)

    def to_dict(self) -> dict[str, object]:
        return {
            "valid": self.valid,
            "findings": [
                {
                    "code": finding.code,
                    "message": finding.message,
                    "location": finding.location,
                    "severity": finding.severity,
                }
                for finding in self.findings
            ],
        }


class WorkPackageSemanticValidator:
    def validate(
        self,
        decomposition: WorkPackageDecompositionDocument,
        *,
        architecture_element_ids: set[str],
        requirement_ids: set[str],
    ) -> WorkPackageValidationReport:
        findings: list[
            WorkPackageValidationFinding
        ] = []

        package_keys = {
            package.package_key
            for package in decomposition.packages
        }

        for package in decomposition.packages:
            for trace in package.architecture_trace:
                if (
                    trace.architecture_element_id
                    not in architecture_element_ids
                ):
                    findings.append(
                        WorkPackageValidationFinding(
                            code=(
                                "UNKNOWN_ARCHITECTURE_ELEMENT"
                            ),
                            message=(
                                f"Package {package.package_key} "
                                "references undefined architecture "
                                "element "
                                f"{trace.architecture_element_id}."
                            ),
                            location=(
                                f"packages."
                                f"{package.package_key}."
                                "architecture_trace"
                            ),
                        )
                    )

                for requirement_id in trace.requirement_ids:
                    if requirement_id not in requirement_ids:
                        findings.append(
                            WorkPackageValidationFinding(
                                code=(
                                    "UNKNOWN_REQUIREMENT"
                                ),
                                message=(
                                    f"Package "
                                    f"{package.package_key} "
                                    "references undefined "
                                    "requirement "
                                    f"{requirement_id}."
                                ),
                                location=(
                                    f"packages."
                                    f"{package.package_key}."
                                    "architecture_trace"
                                ),
                            )
                        )

            for dependency in (
                package.package_dependencies
            ):
                if (
                    dependency.package_key
                    not in package_keys
                ):
                    findings.append(
                        WorkPackageValidationFinding(
                            code=(
                                "UNKNOWN_PACKAGE_DEPENDENCY"
                            ),
                            message=(
                                f"Package {package.package_key} "
                                "depends on undefined package "
                                f"{dependency.package_key}."
                            ),
                            location=(
                                f"packages."
                                f"{package.package_key}."
                                "package_dependencies"
                            ),
                        )
                    )

                if (
                    dependency.package_key
                    == package.package_key
                ):
                    findings.append(
                        WorkPackageValidationFinding(
                            code="SELF_DEPENDENCY",
                            message=(
                                f"Package {package.package_key} "
                                "depends on itself."
                            ),
                            location=(
                                f"packages."
                                f"{package.package_key}."
                                "package_dependencies"
                            ),
                        )
                    )

        if not findings:
            self._assert_acyclic(decomposition)

        return WorkPackageValidationReport(
            valid=not findings,
            findings=findings,
        )

    def _assert_acyclic(
        self,
        decomposition: WorkPackageDecompositionDocument,
    ) -> None:
        graph = {
            package.package_key: {
                dependency.package_key
                for dependency
                in package.package_dependencies
            }
            for package in decomposition.packages
        }

        visiting: set[str] = set()
        visited: set[str] = set()

        def visit(node: str) -> None:
            if node in visiting:
                raise WorkPackageDependencyCycleError(
                    f"Dependency cycle detected at {node}."
                )

            if node in visited:
                return

            visiting.add(node)

            for dependency in graph[node]:
                visit(dependency)

            visiting.remove(node)
            visited.add(node)

        for package_key in graph:
            visit(package_key)

Later increments will add deeper scope, repository, test-command, and execution-limit validation.

from typing import Any


def extract_architecture_element_ids(
    architecture: dict[str, Any],
) -> set[str]:
    collection_names = [
        "decisions",
        "modules",
        "api_contracts",
        "event_contracts",
        "data_entities",
        "interactions",
        "threats",
        "failure_scenarios",
        "deployment_units",
        "open_questions",
    ]

    identifiers: set[str] = set()

    for collection_name in collection_names:
        collection = architecture.get(

            [],
        )

        if not isinstance(collection, list):
            continue

        for item in collection:
            if not isinstance(item, dict):
                continue

            identifier = item.get("id")

            if isinstance(identifier, str):
                identifiers.add(identifier)

    return identifiers


def extract_architecture_requirement_ids(

        "requirement_ids",
        [],
    )

    if not isinstance(requirement_ids, list):
        return set()

    return {
        item
        for item in requirement_ids
        if isinstance(item, str)
    }

Extend WorkPackageService:

    async def complete_decomposition_run(
        self,
        command: CompleteDecompositionRunCommand,
    ) -> list[WorkPackage]:
        run = await self._repository.get_run(
            command.run_id,
            for_update=True,
        )

        if run is None:
            raise WorkPackageRunNotFoundError(
                f"Decomposition run not found: "
                f"{command.run_id}."
            )

        current = WorkPackageDecompositionRunStatus(
            run.status
        )

        target = (
            WorkPackageDecompositionRunStatus.COMPLETED
        )

        assert_run_transition(current, target)

        existing_packages = (
            await self._repository.get_packages_for_run(
                run.id
            )
        )

        if existing_packages:
            raise WorkPackageVersionConflictError(
                "Decomposition run already produced packages."
            )

        package_models: dict[str, WorkPackage] = {}

        for package_payload in command.packages:
            document = WorkPackageDocument.model_validate(
                package_payload
            )

            markdown = render_work_package_markdown(
                document
            )

            checksum = work_package_checksum(
                markdown_content=markdown,
                structured_content=document.model_dump(
                    mode="json"
                ),
            )

            await self._repository.lock_package_version_sequence(
                project_id=run.project_id,
                package_key=document.package_key,
            )

            version = (
                await self._repository.next_package_version(
                    project_id=run.project_id,
                    package_key=document.package_key,
                )
            )

            package = WorkPackage(
                decomposition_run_id=run.id,
                project_id=run.project_id,
                architecture_artifact_id=(
                    run.architecture_artifact_id
                ),
                package_key=document.package_key,
                version=version,
                sequence_number=document.sequence_number,
                title=document.title,
                objective=document.objective,
                status=WorkPackageStatus.DRAFT.value,
                execution_mode=(
                    document.execution_mode.value
                ),
                risk_level=document.risk_level.value,
                markdown_content=markdown,
                structured_content=(
                    document.model_dump(mode="json")
                ),
                content_checksum=checksum,
                estimated_complexity=(
                    document.execution_limits
                    .maximum_changed_files
                ),
                requires_database_migration=(
                    document.requires_database_migration
                ),
                requires_network=(
                    document.requires_network
                ),
                human_review_required=True,
                validation_report=(
                    command.validation_report
                ),
            )

            await self._repository.add_package(package)

            package_models[
                document.package_key
            ] = package

        for package_payload in command.packages:
            document = WorkPackageDocument.model_validate(
                package_payload
            )

            package = package_models[
                document.package_key
            ]

            for dependency_reference in (
                document.package_dependencies
            ):
                dependency_package = package_models[
                    dependency_reference.package_key
                ]

                dependency = WorkPackageDependency(
                    work_package_id=package.id,
                    depends_on_work_package_id=(
                        dependency_package.id
                    ),
                    dependency_type=(
                        dependency_reference
                        .dependency_type.value
                    ),
                    rationale=(
                        dependency_reference.rationale
                    ),
                )

                await self._repository.add_dependency(
                    dependency
                )

        run.status = target.value
        run.validation_report = (
            command.validation_report
        )
        run.finished_at = datetime.now(UTC)

        await self._session.commit()

        return list(package_models.values())
